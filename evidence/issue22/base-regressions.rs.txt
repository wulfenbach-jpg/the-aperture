
    #[test]
    fn issue22_requested_200_percent_text_scale_is_not_silently_clamped() {
        let mut app = four_seat_app();
        let ctx = egui::Context::default();
        app.set_zoom(&ctx, 2.0);
        assert!((app.zoom - 2.0).abs() < f32::EPSILON,
            "requested accessible 200% text scale was clamped to {}", app.zoom);
    }

    #[test]
    fn issue22_a_key_opens_accessibility_controls_without_changing_the_game() {
        let mut app = four_seat_app();
        let before = app.session.game.state.revision;
        press_key(&mut app, 1280.0, 720.0, egui::Key::A);
        let labels = run_frame(&mut app, 1280.0, 720.0, 5);
        assert!(labels.iter().any(|label| label.contains("High contrast")),
            "A must expose the high-contrast control through the real renderer");
        assert_eq!(app.session.game.state.revision, before);
    }
